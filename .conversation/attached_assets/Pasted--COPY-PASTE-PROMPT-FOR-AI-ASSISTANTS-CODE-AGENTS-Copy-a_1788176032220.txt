```sql
  p_workspace_id UUID,
  p_event_type TEXT,
  p_payload JSONB,
  p_trace_id TEXT,
  p_latency_ms INT
) RETURNS UUID AS $$ DECLARE   v_outbox_id UUID; BEGIN   INSERT INTO outbox_events (workspace_id, event_type, payload)   VALUES (p_workspace_id, p_event_type, p_payload)   RETURNING id INTO v_outbox_id;    INSERT INTO audit_logs (     workspace_id, actor_type, event_source, severity, trace_id, latency_ms, payload   ) VALUES (     p_workspace_id, 'webhook', 'whop_webhook', 'info', p_trace_id, p_latency_ms,     jsonb_build_object('outbox_id', v_outbox_id, 'event_type', p_event_type)   );    RETURN v_outbox_id; END; $$ LANGUAGE plpgsql SECURITY DEFINER;

-- Telemetry View
CREATE OR REPLACE VIEW v_system_health AS
SELECT 
  w.id AS workspace_id,
  w.name AS workspace_name,
  (SELECT COUNT(*) FROM outbox_events WHERE workspace_id = w.id AND processed = FALSE) AS unprocessed_outbox_count,
  (SELECT COUNT(*) FROM bot_events WHERE workspace_id = w.id AND status = 'pending') AS pending_bot_queue_depth,
  (SELECT COUNT(*) FROM failed_events WHERE workspace_id = w.id) AS total_dlq_failures,
  (SELECT AVG(latency_ms)::INT FROM audit_logs WHERE workspace_id = w.id AND created_at > NOW() - INTERVAL '1 hour') AS avg_webhook_latency_1h
FROM workspaces w;

-- Expiry State Machine Function
CREATE OR REPLACE FUNCTION check_and_expire_subscriptions() 
RETURNS VOID AS $$ BEGIN   UPDATE members    SET status = 'grace'   WHERE status = 'active' AND expiry_date < NOW();    UPDATE members    SET status = 'churned'   WHERE status = 'grace' AND expiry_date < (NOW() - INTERVAL '48 hours'); END; $$ LANGUAGE plpgsql SECURITY DEFINER;
```

## 2. API WEBHOOK ROUTE (app/api/webhooks/whop/route.ts)
```typescript
import { NextResponse } from 'next/server';
import crypto from 'crypto';
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

export async function POST(req: Request) {
  const startTime = Date.now();
  const traceId = crypto.randomUUID();

  try {
    const rawBody = await req.text();
    const signature = req.headers.get('webhook-signature') || req.headers.get('x-whop-signature');
    const webhookSecret = process.env.WHOP_WEBHOOK_SECRET!;

    if (!signature) {
      return NextResponse.json({ error: 'Missing HMAC signature' }, { status: 401 });
    }

    const hmac = crypto.createHmac('sha256', webhookSecret).update(rawBody).digest('base64');
    const expectedSignature = `v1,${hmac}`;

    if (!crypto.timingSafeEqual(Buffer.from(signature), Buffer.from(expectedSignature))) {
      return NextResponse.json({ error: 'Invalid HMAC Signature' }, { status: 401 });
    }

    const payload = JSON.parse(rawBody);
    const eventType = payload.action || payload.event || 'payment.succeeded';
    const workspaceId = payload.data?.custom_fields?.workspace_id || payload.workspace_id;
    const idempotencyKey = payload.id || `whop_${traceId}`;

    if (!workspaceId) {
      return NextResponse.json({ error: 'Missing workspace_id context' }, { status: 400 });
    }

    const { data: outboxId, error: rpcError } = await supabase.rpc('process_webhook_ingestion_atomic', {
      p_workspace_id: workspaceId,
      p_event_type: eventType,
      p_payload: { ...payload, idempotency_key: idempotencyKey },
      p_trace_id: traceId,
      p_latency_ms: Date.now() - startTime
    });

    if (rpcError) throw rpcError;

    return NextResponse.json({ status: 'queued', outbox_id: outboxId, trace_id: traceId }, { status: 200 });
  } catch (err: any) {
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}
```

## 3. OUTBOX PROCESSOR EDGE FUNCTION (supabase/functions/outbox-processor/index.ts)
```typescript
import { createClient } from '[https://esm.sh/@supabase/supabase-js@2](https://esm.sh/@supabase/supabase-js@2)';

Deno.serve(async () => {
  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  );

  const { data: events, error } = await supabase
    .from('outbox_events')
    .select('*')
    .eq('processed', false)
    .order('created_at', { ascending: true })
    .limit(20);

  if (error || !events) return new Response(JSON.stringify({ error }), { status: 500 });

  for (const evt of events) {
    const { workspace_id, payload } = evt;
    const telegramId = payload.data?.telegram_id || payload.user_id;
    const grossAmount = payload.data?.final_amount || payload.amount || 0;
    const feeAmount = payload.data?.fee || 0;
    const idempotencyKey = payload.idempotency_key;

    const expiryDate = new Date();
    expiryDate.setDate(expiryDate.getDate() + 30);

    const { data: member, error: memberError } = await supabase
      .from('members')
      .upsert({
        workspace_id,
        telegram_id: telegramId,
        username: payload.data?.username || null,
        status: 'active',
        expiry_date: expiryDate.toISOString(),
        deleted_at: null
      }, { onConflict: 'workspace_id,telegram_id' })
      .select()
      .single();

    if (!memberError && member) {
      await supabase.from('payments').insert({
        workspace_id,
        member_id: member.id,
        idempotency_key: idempotencyKey,
        gross_amount: grossAmount,
        fee_amount: feeAmount,
        currency: 'USD',
        status: 'confirmed'
      });

      await supabase.from('bot_events').insert({
        workspace_id,
        telegram_id: telegramId,
        event_type: 'invite',
        status: 'pending',
        retry_count: 0
      });
    }

    await supabase.from('outbox_events').update({ processed: true }).eq('id', evt.id);
  }

  return new Response(JSON.stringify({ processed: events.length }), { status: 200 });
});
```

```javascript
const { createClient } = require('@supabase/supabase-js');
const TelegramBot = require('node-telegram-bot-api');

const supabase = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_SERVICE_ROLE_KEY
);

async function isWorkspaceThrottled(workspaceId) {
  const { data } = await supabase
    .from('rate_limit_state')
    .select('blocked_until')
    .eq('workspace_id', workspaceId)
    .single();

  if (!data || !data.blocked_until) return false;
  return new Date(data.blocked_until) > new Date();
}

async function setWorkspaceThrottle(workspaceId, seconds) {
  const blockedUntil = new Date(Date.now() + seconds * 1000).toISOString();
  await supabase
    .from('rate_limit_state')
    .upsert({
      workspace_id: workspaceId,
      blocked_until: blockedUntil,
      last_request_at: new Date().toISOString()
    });
}

async function processBotQueue() {
  const { data: events, error } = await supabase
    .from('bot_events')
    .select('*, workspaces(telegram_bot_token)')
    .eq('status', 'pending')
    .lt('retry_count', 5)
    .limit(10);

  if (error || !events || events.length === 0) return;

  for (const evt of events) {
    if (await isWorkspaceThrottled(evt.workspace_id)) {
      console.log(`[Global Rate Limit] Workspace ${evt.workspace_id} throttled. Skipping.`);
      continue;
    }

    const token = evt.workspaces?.telegram_bot_token || process.env.DEFAULT_TELEGRAM_BOT_TOKEN;
    const bot = new TelegramBot(token, { polling: false });

    try {
      await supabase.from('bot_events').update({ status: 'processing' }).eq('id', evt.id);

      if (evt.event_type === 'invite') {
        const link = await bot.createChatInviteLink(evt.telegram_id, { member_limit: 1 });
        await bot.sendMessage(evt.telegram_id, `Welcome! Here is your access link: ${link.invite_link}`);
      } else if (evt.event_type === 'remove') {
        await bot.banChatMember(evt.telegram_id, evt.telegram_id);
        await bot.unbanChatMember(evt.telegram_id, evt.telegram_id);
      }

      await supabase.from('bot_events').update({ status: 'completed', updated_at: new Date() }).eq('id', evt.id);

    } catch (err) {
      const is429 = err.response && err.response.statusCode === 429;
      const newRetryCount = evt.retry_count + 1;

      if (is429) {
        const retryAfter = err.response.body?.parameters?.retry_after || 30;
        await setWorkspaceThrottle(evt.workspace_id, retryAfter);
      }

      if (newRetryCount >= 5) {
        await supabase.from('failed_events').insert({
          workspace_id: evt.workspace_id,
          original_event_id: evt.id,
          event_source: 'bot_queue',
          payload: evt,
          failure_reason: err.message || 'Exceeded 5 retries'
        });

        await supabase.from('bot_events').update({ status: 'failed', last_error: err.message }).eq('id', evt.id);
      } else {
        await supabase.from('bot_events').update({
          status: 'pending',
          retry_count: newRetryCount,
          last_error: err.message,
          updated_at: new Date()
        }).eq('id', evt.id);
      }
    }
  }
}

async function startWorker() {
  console.log('🤖 Telegram Bot Queue Worker Live...');
  while (true) {
    try {
      await processBotQueue();
    } catch (e) {
      console.error('[Worker Loop Error]', e);
    }
    await new Promise((r) => setTimeout(r, 2000));
  }
}

startWorker();
```
